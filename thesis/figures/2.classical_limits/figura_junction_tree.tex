    \node[original] (A3) at (1, -1.5) {$X_3$};

    \draw[arrow] (A1) -- (A2);
    \draw[arrow] (A1) -- (A3);
    \draw[arrow] (A2) -- (A4);
    \draw[arrow] (A3) -- (A4);

    % Panel B: Grafo Moralizado
    \node[draw=none] (titleB) at (4, 1.5) {\textbf{b) Grafo Moralizado}};
    \node[moral] (B1) at (4, 0) {$X_1$};

    \node[draw=none] (titleC) at (9, 1.5) {\textbf{c) Junction Tree (Cliques)}};

    \caption{\textbf{Proceso de Compilación Clásica (Junction Tree) y Explosión del Treewidth.} \textbf{a)} Una red causal con variables interdependientes (ej. reacciones fotoquímicas). \textbf{b)} La moralización requiere "casar" a los padres ($X_2, X_3$) del nodo hijo ($X_4$), introduciendo enlaces no dirigidos de relleno (\textit{fill-in edges}). \textbf{c)} La triangulación resultante agrupa las variables en grandes subconjuntos llamados \textit{cliques}. El coste de memoria escala exponencialmente con el tamaño del clique más grande ($w = 2$ en este esquema simplificado), provocando la saturación de RAM en redes reales.}
